rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    
    // ==========================================
    // 1. Life Tracker 生活儀表板的專屬規則
    // ==========================================
    match /LifeApp_Records/{docId} {
      allow read, delete: if request.auth != null && resource.data.uid == (request.auth.uid + "_life");
      allow create, update: if request.auth != null && request.resource.data.uid == (request.auth.uid + "_life");
    }
    match /LifeApp_Settings/{settingId} {
      allow read, write: if request.auth != null && settingId == (request.auth.uid + "_life");
    }

    // ==========================================
    // 2. 楓之谷系統的專屬規則
    // ==========================================
    match /DropRecords/{docId} {
      allow read, delete: if request.auth != null && resource.data.uid == request.auth.uid;
      allow create, update: if request.auth != null && request.resource.data.uid == request.auth.uid;
    }
    match /UserSettings/{userId} {
      allow read, write: if request.auth != null && userId == request.auth.uid;
    }

    // ==========================================
    // 3. 戰術白板系統 2.0 的專屬規則
    // ==========================================
    match /Whiteboard_Data/{userId} {
      allow read, write: if request.auth != null && userId == request.auth.uid;
    }

    // ==========================================
    // 4. Lite-Pocket 知識庫系統的專屬規則
    // ==========================================
    match /LitePocket_Data/{userId} {
      allow read, write: if request.auth != null && userId == request.auth.uid;
    }

    // ==========================================
    // 5. Sandwich 圖文記憶學習系統的專屬規則
    // ==========================================
    match /Sandwich_Data/{userId} {
      allow read, write: if request.auth != null && userId == request.auth.uid;
    }

    // ==========================================
    // 6. 影片筆記工作站的專屬規則
    // ==========================================
    match /VideoNotes_Data/{userId} {
      allow read, write: if request.auth != null && userId == request.auth.uid;
    }

    // ==========================================
    // 7. 部落格動態模板產生器
    // ==========================================
    match /Dynamic_Templates/{userId} {
      allow read, write: if request.auth != null && userId == request.auth.uid;
    }

    // ==========================================
    // 8. Project Lab 專案檢討系統
    // ==========================================
    match /ProjectLab_Data/{userId} {
      allow read, write: if request.auth != null && userId == request.auth.uid;
    }
  }
}
